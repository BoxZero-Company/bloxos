===== GUI FRAMEBUFFER =====

pub struct Framebuffer<'a> {
    pub buffer: &'a mut [u32],
    pub width: usize,
    pub height: usize,
}

impl<'a> Framebuffer<'a> {
    pub fn new(
        buffer: &'a mut [u32],
        width: usize,
        height: usize,
    ) -> Self {
        Self {
            buffer,
            width,
            height,
        }
    }

    pub fn clear(&mut self, color: u32) {
        self.buffer.fill(color);
    }

    pub fn pixel(&mut self, x: usize, y: usize, color: u32) {
        if x < self.width && y < self.height {
            let index = y * self.width + x;
            self.buffer[index] = color;
        }
    }

    pub fn rectangle(
        &mut self,
        x: usize,
        y: usize,
        width: usize,
        height: usize,
        color: u32,
    ) {
        let end_x = x.saturating_add(width).min(self.width);
        let end_y = y.saturating_add(height).min(self.height);

        for py in y.min(self.height)..end_y {
            for px in x.min(self.width)..end_x {
                self.pixel(px, py, color);
            }
        }
    }
}

    }

#[derive(Debug, Clone, Copy, PartialEq)]
pub enum InstallStep {
    Welcome,
    Language,
    Account,
    Purpose,
    Network,
    Partition,
    Summary,
    Installing,
    Complete,
}

fn main() {
    println!("BloxOS Installer");
    println!("Version: 0.1.0");
    println!("Installer initialization placeholder");
}

pub struct Installer {
    pub step: InstallStep,
    pub language: String,
    pub username: String,
    pub password: String,
    pub confirm_password: String,
    pub recovery: String,
}

impl Installer {
    pub fn new() -> Self {
        Self {
            step: InstallStep::Welcome,
            language: "English".into(),
            username: String::new(),
            password: String::new(),
            confirm_password: String::new(),
            recovery: String::new(),
        }
    }

    pub fn next(&mut self) -> Result<(), &'static str> {
        self.step = match self.step {
            InstallStep::Welcome => InstallStep::Language,
            InstallStep::Language => InstallStep::Account,
            InstallStep::Account => {
                if self.username.trim().is_empty()
                    || self.password.is_empty()
                    || self.confirm_password.is_empty()
                    || self.recovery.trim().is_empty()
                {
                    return Err("All account fields are required");
                }

                if self.password != self.confirm_password {
                    return Err("Passwords do not match");
                }

                InstallStep::Purpose
            }
            InstallStep::Purpose => InstallStep::Network,
            InstallStep::Network => InstallStep::Partition,
            InstallStep::Partition => InstallStep::Summary,
            InstallStep::Summary => InstallStep::Installing,
            InstallStep::Installing => InstallStep::Complete,
            InstallStep::Complete => InstallStep::Complete,
        };

        Ok(())
    }
}

pub struct LockScreen {
    pub locked: bool,
    pub username: String,
}

impl LockScreen {
    pub fn new(username: String) -> Self {
        Self {
            locked: true,
            username,
        }
    }

    pub fn unlock(&mut self, authenticated: bool) -> bool {
        if authenticated {
            self.locked = false;
        }

        !self.locked
    }
}
